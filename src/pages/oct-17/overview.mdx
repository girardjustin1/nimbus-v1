export const protoUrl = (route) =>
  (typeof window !== "undefined" && window.location.hostname === "localhost" ? "http://localhost:5190/das/v4/" : "./das/v4/") + `#/${route}`;
export const storyUrl = (id) => `./?path=/story/${id}`;

export const Pill = ({ href, kind, children }) => (
  <a href={href} target="_top" style={{
    display: "inline-block", whiteSpace: "nowrap", textDecoration: "none", fontSize: 14, fontWeight: 600, padding: "2px 9px", borderRadius: 999,

  }}>{children}</a>
);

export const LinkRow = ({ label, children }) => (
  <div style={{ display: "flex", flexWrap: "wrap", alignItems: "center", gap: 6, margin: "2px 0" }}>
    <span style={{ fontSize: 13, fontWeight: 700, color: "#667085", textTransform: "uppercase", letterSpacing: 0.4, minWidth: 82 }}>{label}</span>
    {children}
  </div>
);

export const Where = ({ slug }) => {
  const l = links[slug];
  if (!l) return null;
  return (
    <div>
      <LinkRow label="Prototype">{l.prototype.map(([t, r]) => <Pill key={t} kind="proto" href={protoUrl(r)}>{t} ↗</Pill>)}</LinkRow>
      <LinkRow label="Storybook">{l.stories.map(([t, id]) => <Pill key={t} kind="story" href={storyUrl(id)}>{t} →</Pill>)}</LinkRow>
    </div>
  );
};

export const Chip = ({ children, tone }) => (
  <span style={{
    background: tone === "Applied" || tone === "Decided" ? "#E6F6F6" : tone === "Concepts" ? "#FFF4E5" : "#FBE9F1",

    borderRadius: 999, padding: "1px 8px", fontSize: 15, fontWeight: 700, whiteSpace: "nowrap",
  }}>{children}</span>
);

## Charter check: keywords

- **Matching.** The charter's setup wireframe had an ANY / ALL toggle defaulting to ANY;
  it was dropped on 1 Oct. Product confirmed on 19 Oct that keywords match on **ANY**,
  like geos, with no toggle.
- **Reporting.** The charter's guardrail keeps per-keyword breakdowns to CSV and the API
  at launch, with no per-keyword charting in the dashboard. Full performance currently
  shows keywords on screen; the CSV-only version is concept B. Waiting on Product.
- **Counting.** With ANY, one impression can match several keywords. It counts under
  the first it matched, so per-keyword rows add up to the campaign's total.

## Campaign Health maths is a placeholder

Product is settling the rule with engineering, and said it can run in the front end.
Until then the prototype uses: **green** when spend is within 10% of the share of the
flight that has gone (ahead counts as green); **orange** when it's behind but the daily
spend needed to finish is at most 1.5× today's; **red** beyond that. Only live campaigns
with a budget have a health. It is written down once, in
`prototypes/das/v4/screens/campaign-data.ts`, so it can be replaced in one place.
