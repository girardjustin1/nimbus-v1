export const protoUrl = (route) =>
  (typeof window !== "undefined" && /^(localhost|127\.0\.0\.1)$/.test(window.location.hostname) ? "http://localhost:5190/das/v6/" : "./das/v6/") + `#/${route}`;
export const storyUrl = (id) => (id.endsWith("--docs") ? `./?path=/docs/${id}` : `./?path=/story/${id}`);

export const Pill = ({ href, kind, children }) => (
  <a href={href} target="_top" style={{
    display: "inline-block", whiteSpace: "nowrap", textDecoration: "none", fontSize: 15, fontWeight: 600, padding: "2px 9px", borderRadius: 999,
    color: kind === "proto" ? "#A94579" : "#1F7F80",
    background: kind === "proto" ? "#FCE7F1" : "#E6F6F6",
  }}>{children}</a>
);

export const LinkRow = ({ label, children }) => (
  <div style={{ display: "flex", flexWrap: "wrap", alignItems: "center", gap: 6, margin: "2px 0" }}>
    <span style={{ fontSize: 15, fontWeight: 700, color: "#667085", textTransform: "uppercase", letterSpacing: 0.4, minWidth: 92 }}>{label}</span>
    {children}
  </div>
);

export const Where = ({ slug }) => {
  const l = links[slug];
  if (!l) return null;
  return (
    <div>
      {l.prototype.length > 0 && <LinkRow label="Prototype">{l.prototype.map(([t, r]) => <Pill key={t} kind="proto" href={protoUrl(r)}>{t} ↗</Pill>)}</LinkRow>}
      {l.stories.length > 0 && <LinkRow label="Storybook">{l.stories.map(([t, id]) => <Pill key={t} kind="story" href={storyUrl(id)}>{t} →</Pill>)}</LinkRow>}
    </div>
  );
};

**First batch, recommendation: C.** A and B are the most literal answer, but two scales on one chart
make the reader check which axis each line belongs to, and the lines cross wherever the
two scales happen to meet. C keeps the pacing chart as Product and sales saw it and adds
opportunity beneath it on the same days. D is the strongest if one picture matters more
than the cumulative pacing line. The overview opens on C; the **Chart** menu in the prototype toolbar switches any campaign overview between Round 6 and A–I.

**On one chart, recommendation: G or I.** Both keep one visible scale, so there's nothing
to match up. G shows the shape of every day's opportunity behind pacing; I says in words
why pacing slowed and leaves ordinary days alone. F is the overlay as asked, made easier to
read, but it still has two scales.
